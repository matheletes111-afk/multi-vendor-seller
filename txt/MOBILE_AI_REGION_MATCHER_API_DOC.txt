================================================================================
MOBILE & WEB API DOCUMENTATION: AI-POWERED DELIVERY REGION & ZONE MATCHER
================================================================================

API ENDPOINT:
  POST /api/customer/checkout/ai-match-region

SUMMARY:
  This API accepts a customer's physical delivery address (street, city, state/province, landmarks)
  and leverages a multi-tier spatial resolution engine with strict priority order:
    1. Priority 1 (Highest): Specific Delivery Zones (ZONE 1 to ZONE 15) evaluated from streets,
       neighborhoods, landmarks, and area names.
    2. Priority 2: Gemini AI spatial intelligence prioritizing specific delivery zones (ZONE 1 to 15).
    3. Priority 3: Administrative Provinces (Western Area, Northern Province, etc.) used ONLY if
       the address does not belong to any specific delivery zone.
    4. Priority 4 (Lowest): "Other" fallback.
  It also returns the exact live delivery surcharge (in NLe) configured in the database admin settings.

NOTE FOR MOBILE DEVELOPERS:
  YES! This exact same API is used for both Next.js Web Checkout and Flutter Mobile Apps.
  There is zero separate logic required for mobile apps.

===============================================================================
1. REQUEST SPECIFICATION
===============================================================================

HTTP METHOD:
  POST

FULL URL:
  http://<YOUR_DOMAIN_OR_LOCAL_IP>:3000/api/customer/checkout/ai-match-region

HEADERS:
  Content-Type: application/json
  (Optional) Cookie: authjs.session-token=... (or Authorization: Bearer <token>)

REQUEST BODY SCHEMA (JSON):
{
  "addressLine1": "15 Beach Road, near Lakka Junction",
  "addressLine2": "Apt 3B",
  "city": "Freetown",
  "state": "Western Area",
  "postalCode": "90001"
}

FIELD DETAILS:
  - addressLine1 (string, optional): Primary street address or landmark (e.g. "15 Beach Road")
  - addressLine2 (string, optional): Apartment, suite, or building name
  - city         (string, optional): City or town name (e.g. "Freetown", "Bo", "Makeni")
  - state        (string, optional): State / Province / District / Area name (e.g. "Western Area")
  - postalCode   (string, optional): Zip or Postal code

===============================================================================
2. RESPONSE SPECIFICATIONS
===============================================================================

2.1 SUCCESS RESPONSE (200 OK):
{
  "success": true,
  "data": {
    "matchedRegion": "LAKKA",
    "zone": "ZONE 1",
    "confidence": 0.98,
    "reasoning": "Address specifies Lakka Beach Road and Lakka Junction in Western Area.",
    "isAiMatched": true,
    "charge": 50
  }
}

RESPONSE FIELD DEFINITIONS:
  - matchedRegion (string):  Exact region matched from system list (e.g. "LAKKA", "CONGO CROSS", "PZ", "Western Area", "Other").
  - zone          (string):  Parent zone name (e.g. "ZONE 1", "ZONE 4", "ZONE 5", "Administrative Provinces", "Other").
  - confidence    (number):  Confidence score 0.0 to 1.0 (e.g. 0.98 = 98%).
  - reasoning     (string):  Short spatial explanation from AI/matcher.
  - isAiMatched   (boolean): true = matched by Gemini AI. false = deterministic exact match or fallback.
  - charge        (number):  Live delivery surcharge fee in NLe from Admin Settings (e.g. 50.0).

2.2 FALLBACK / UNKNOWN LOCATION RESPONSE (200 OK):
{
  "success": true,
  "data": {
    "matchedRegion": "Other",
    "zone": "Other",
    "confidence": 0.5,
    "reasoning": "Standard fallback region.",
    "isAiMatched": false,
    "charge": 50
  }
}

2.3 ERROR RESPONSE (400 / 500):
{
  "success": false,
  "error": "Invalid address input payload"
}

class AIRegionMatchResult {
  final String matchedRegion;
  final String zone;
  final double confidence;
  final String reasoning;
  final bool isAiMatched;
  final double charge;

  const AIRegionMatchResult({
    required this.matchedRegion,
    required this.zone,
    required this.confidence,
    required this.reasoning,
    required this.isAiMatched,
    required this.charge,
  });

  factory AIRegionMatchResult.fromJson(Map<String, dynamic> json) {
    return AIRegionMatchResult(
      matchedRegion: json['matchedRegion']?.toString() ?? 'Other',
      zone: json['zone']?.toString() ?? 'Other',
      confidence: (json['confidence'] is num)
          ? (json['confidence'] as num).toDouble()
          : 0.5,
      reasoning: json['reasoning']?.toString() ?? '',
      isAiMatched: json['isAiMatched'] == true,
      charge: (json['charge'] is num)
          ? (json['charge'] as num).toDouble()
          : 0.0,
    );
  }

  Map<String, dynamic> toJson() => {
        'matchedRegion': matchedRegion,
        'zone': zone,
        'confidence': confidence,
        'reasoning': reasoning,
        'isAiMatched': isAiMatched,
        'charge': charge,
      };

  /// Clean display string helper for UI (e.g. "PZ (ZONE 5)", "Western Area", "Other")
  String get displayLabel {
    if (zone == 'Administrative Provinces' ||
        matchedRegion == 'Other' ||
        matchedRegion == zone) {
      return matchedRegion;
    }
    return '$matchedRegion ($zone)';
  }
}

import 'dart:convert';
import 'package:flutter/foundation.dart';
import 'package:http/http.dart' as http;
import '../models/ai_region_match_result.dart';

class AIRegionApiService {
  // TODO: Replace with your actual production domain
  static const String baseUrl = 'https://your-domain.com';
  static const Duration _timeout = Duration(seconds: 15);

  /// Calls the AI region matcher API and returns the resolved region/zone/charge.
  /// Returns null on failure or network error — always handle null in the caller.
  static Future<AIRegionMatchResult?> matchAddressRegion({
    required String addressLine1,
    String? addressLine2,
    required String city,
    required String state,
    String? postalCode,
    String? sessionCookie,
  }) async {
    try {
      final uri = Uri.parse('$baseUrl/api/customer/checkout/ai-match-region');

      final headers = <String, String>{
        'Content-Type': 'application/json',
      };
      if (sessionCookie != null && sessionCookie.isNotEmpty) {
        headers['Cookie'] = sessionCookie;
      }

      final body = jsonEncode({
        'addressLine1': addressLine1,
        'addressLine2': addressLine2 ?? '',
        'city': city,
        'state': state,
        'postalCode': postalCode ?? '',
      });

      final response = await http
          .post(uri, headers: headers, body: body)
          .timeout(_timeout);

      if (response.statusCode == 200) {
        final decoded = jsonDecode(response.body);
        if (decoded is Map<String, dynamic> &&
            decoded['success'] == true &&
            decoded['data'] is Map<String, dynamic>) {
          return AIRegionMatchResult.fromJson(
            decoded['data'] as Map<String, dynamic>,
          );
        }
      }
    } on Exception catch (e) {
      debugPrint('AIRegionApiService error: $e');
    }
    return null;
  }
}

// Inside your checkout screen StatefulWidget _CheckoutScreenState:

AIRegionMatchResult? _aiRegionResult;
bool _isAiLoading = false;

Future<void> _resolveRegion(Address selectedAddress) async {
  setState(() => _isAiLoading = true);
  final result = await AIRegionApiService.matchAddressRegion(
    addressLine1: selectedAddress.line1,
    addressLine2: selectedAddress.line2,
    city: selectedAddress.city,
    state: selectedAddress.state,
    postalCode: selectedAddress.postalCode,
  );
  setState(() {
    _aiRegionResult = result;
    _isAiLoading = false;
  });
}

// In your widget build method, under the selected delivery address card:
if (_isAiLoading)
  const Padding(
    padding: EdgeInsets.symmetric(vertical: 6),
    child: Row(
      children: [
        SizedBox(width: 14, height: 14, child: CircularProgressIndicator(strokeWidth: 2)),
        SizedBox(width: 8),
        Text('Determining delivery zone…', style: TextStyle(fontSize: 11, color: Color(0xFF0F766E))),
      ],
    ),
  )
else if (_aiRegionResult != null)
  AIRegionBadgeWidget(aiResult: _aiRegionResult!)

// In your order summary (Delivery Breakdown card):
Text(
  'Regional Surcharge (${_aiRegionResult?.displayLabel ?? "Other"}): '
  'NLe ${(_aiRegionResult?.charge ?? 0.0).toStringAsFixed(2)}',
)
